\begin{frame}{A focused workflow}
  \begin{enumerate}
    \item Identify one message for each slide.
    \item Support it with readable evidence.
    \item Rehearse the transition to the next idea.
  \end{enumerate}
\end{frame}

\begin{frame}{A small mathematical example}
  For \(f(x)=x^2\), the rate of change is
  \[
    f'(x)=2x.
  \]
  \begin{block}{Interpretation}
    At \(x=1\), the derivative equals \(2\).
  \end{block}
\end{frame}

\begin{frame}{Takeaway}
  A compiled slide is only a starting point. Check contrast, font size,
  reading order, and whether the message makes sense without narration.
\end{frame}
